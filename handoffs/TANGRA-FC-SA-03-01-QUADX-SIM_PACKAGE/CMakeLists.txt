cmake_minimum_required(VERSION 3.16)
project(tangra_fc_step3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tangra_fc_quadx_sim src/quadx_simulation.cpp)
target_include_directories(tangra_fc_quadx_sim PUBLIC include)
add_executable(test_quadx_sim tests/test_quadx_simulation.cpp)
target_link_libraries(test_quadx_sim PRIVATE tangra_fc_quadx_sim)
add_executable(test_real_fc_adapter_contracts tests/test_real_fc_adapter_contracts.cpp)
target_link_libraries(test_real_fc_adapter_contracts PRIVATE tangra_fc_quadx_sim)
enable_testing()
add_test(NAME quadx_sim COMMAND test_quadx_sim)
add_test(NAME real_fc_adapter_contracts COMMAND test_real_fc_adapter_contracts)
