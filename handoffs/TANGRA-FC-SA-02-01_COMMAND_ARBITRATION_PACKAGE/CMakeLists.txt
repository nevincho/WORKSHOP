cmake_minimum_required(VERSION 3.16)
project(tangra_fc_step2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tangra_fc_command_arbitration src/command_arbitration.cpp)
target_include_directories(tangra_fc_command_arbitration PUBLIC include)
add_executable(test_command_arbitration tests/test_command_arbitration.cpp)
target_link_libraries(test_command_arbitration PRIVATE tangra_fc_command_arbitration)
enable_testing()
add_test(NAME command_arbitration COMMAND test_command_arbitration)
